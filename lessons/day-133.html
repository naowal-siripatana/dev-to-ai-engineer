import contextvars
import logging
import re
import uuid

request_id_var: contextvars.ContextVar[str] = contextvars.ContextVar("request_id", default="-")

# What a request id may look like when it comes from outside. Anything else is replaced: a header is user input, and a newline in it
# would let a caller forge extra lines in your log.
VALID_REQUEST_ID = re.compile(r"[A-Za-z0-9._-]{1,64}")


def clean_request_id(value: str | None) -> str:
    """Use the id the caller sent if it is safe, otherwise make a new one."""
    if value is not None and VALID_REQUEST_ID.fullmatch(value):
        return value
    return uuid.uuid4().hex


class RequestIdFilter(logging.Filter):
    """Puts the id of the current request on every log record, so that one grep follows a request through the whole system."""

    def filter(self, record: logging.LogRecord) -> bool:
        record.request_id = request_id_var.get()
        return True


LOG_FORMAT = "%(asctime)s %(levelname)s %(name)s rid=%(request_id)s %(message)s"


def setup_logging() -> None:
    logging.basicConfig(level=logging.INFO, format=LOG_FORMAT)
    for handler in logging.getLogger().handlers:
        handler.addFilter(RequestIdFilter())

Step 2: the table and the store

src/my_ai_project/metrics.py:

import logging
import os
from dataclasses import dataclass
from typing import Protocol

import psycopg

logger = logging.getLogger("metrics")

# EXAMPLE prices in US dollars per million tokens. They are placeholders: put the real prices of YOUR model here, from its price list.
PRICES_PER_MILLION = {"input": 1.00, "output": 5.00}


def cost_usd(input_tokens: int | None, output_tokens: int | None, prices: dict[str, float] | None = None) -> float | None:
    """None means 'not measured'. 0.0 means 'measured, and it was free' (for example: no model was called)."""
    if input_tokens is None or output_tokens is None:
        return None
    price = prices or PRICES_PER_MILLION
    return (input_tokens * price["input"] + output_tokens * price["output"]) / 1_000_000


@dataclass
class RequestMetrics:
    request_id: str
    route: str
    role: str
    status_code: int
    latency_ms: float
    outcome: str                 # answered, refused, error
    model: str
    input_tokens: int | None     # None = not measured
    output_tokens: int | None


class Metrics(Protocol):
    def record(self, row: RequestMetrics) -> None: ...


SCHEMA = """
CREATE TABLE IF NOT EXISTS requests (
    id bigserial PRIMARY KEY,
    time timestamptz NOT NULL DEFAULT now(),
    request_id text NOT NULL,
    route text NOT NULL,
    role text NOT NULL,
    status_code int NOT NULL,
    latency_ms real NOT NULL,
    outcome text NOT NULL,
    model text NOT NULL,
    input_tokens int,
    output_tokens int,
    cost_usd numeric(12, 6)
);
CREATE INDEX IF NOT EXISTS requests_time_idx ON requests (time);
CREATE INDEX IF NOT EXISTS requests_request_id_idx ON requests (request_id);
"""


class MetricsStore:
    """One row per request in PostgreSQL: how long it took, what it cost, how it ended."""

    def __init__(self, dsn: str) -> None:
        self.dsn = dsn
        self.ready = False

    def record(self, row: RequestMetrics) -> None:
        """Never raises: if the database is down, the metrics are lost, but the user's request must not be."""
        try:
            with psycopg.connect(self.dsn) as conn:
                if not self.ready:
                    conn.execute(SCHEMA)
                    self.ready = True
                conn.execute(
                    "INSERT INTO requests (request_id, route, role, status_code, latency_ms, outcome, model, input_tokens, output_tokens, cost_usd) "
                    "VALUES (%s, %s, %s, %s, %s, %s, %s, %s, %s, %s)",
                    (row.request_id, row.route, row.role, row.status_code, row.latency_ms, row.outcome, row.model,
                     row.input_tokens, row.output_tokens, cost_usd(row.input_tokens, row.output_tokens)),
                )
        except psycopg.Error as err:
            logger.warning("could not store the metrics of %s: %s", row.request_id, err)


def default_store() -> MetricsStore:
    """The metrics go to the same database, unless METRICS_DATABASE_URL says otherwise."""
    return MetricsStore(os.environ.get("METRICS_DATABASE_URL") or os.environ.get("DATABASE_URL", "postgresql://assistant:assistant@localhost:5433/assistant"))

Step 5: the five numbers

One query. Define each number before you show it:

DEFINE EVERY NUMBER BEFORE YOU SHOW IT
success rate   (answered + refused) / all requests.   A refusal is a result, not an error; an error is a failure to give a result.
failed         outcome = 'error' (status 4xx or 5xx). Split 4xx (the caller) from 5xx (you) when it matters.
average time   with the median beside it, and the worst case. Say whether the cold start is included.
total cost     the sum of the estimated cost, with the source of the tokens and the date of the prices written next to it.
-- The five numbers of the Module 12 dashboard, from the requests table. Only /api/ask rows.
SELECT
    count(*)                                                                    AS questions,
    round(100.0 * count(*) FILTER (WHERE outcome IN ('answered', 'refused')) / count(*), 1) AS success_percent,
    round(avg(latency_ms)::numeric, 0)                                          AS avg_ms,
    round((percentile_cont(0.5) WITHIN GROUP (ORDER BY latency_ms))::numeric, 0) AS median_ms,
    round(coalesce(sum(cost_usd), 0), 6)                                        AS total_cost_usd,
    count(*) FILTER (WHERE outcome = 'error')                                   AS failed
FROM requests
WHERE route = '/api/ask';

5 · Stuck? anytime

  • The ids differ between the logs: the gateway must send the id (check the client), and the Python middleware must read the header x-request-id.
  • The table is empty: look for the warning “could not store the metrics” in the Python log: the database address may be wrong.
  • The success rate is 100% and you know something failed: a refusal is not an error, but a 4xx must be an error row. Check the outcome logic in the middleware.
  • The cost is NULL: tokens are not measured for that route (the tool loop does not report tokens yet).
  • A request that the gateway rejects (400) has no row: it never reached Python. If you want it, record in the gateway too.

7 · Check yourself 5 min

  1. Why does the average latency of 187 ms hide the truth?
  2. Why is a refusal counted as a success in the success rate?
  3. Why write “estimate” and “example prices” next to the cost?
Answers
  1. One request (the cold start, 3584 ms) pulls it up; the median is 8 ms. Show the median and the worst case too.
  2. A refusal is a correct result (the assistant answered that it does not know). A failure is when no result is given at all.
  3. The tokens were estimated with a tokenizer that is not the model’s own, and the prices are placeholders. A cost number without its source can mislead.
