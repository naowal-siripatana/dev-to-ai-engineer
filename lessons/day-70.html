Step 2: the route

Add these to api.py. Literal and the limits make FastAPI check the parameters for you:

class SearchHit(BaseModel):
    citation: str
    doc_id: str
    chunk_index: int
    title: str
    kind: str
    text: str
    similarity: float


class SearchResponse(BaseModel):
    question: str
    kind: str | None
    results: list[SearchHit]


@app.get("/api/search")
def search(
    q: Annotated[str, Query(min_length=1, max_length=500)],
    embedder: EmbedderDep,
    store: StoreDep,
    top_k: Annotated[int, Query(ge=1, le=10)] = 3,
    kind: Literal["manual", "faq", "error"] | None = None,
) -> SearchResponse:
    try:
        found = retrieve(q, embedder, store, top_k=top_k, kind=kind)
    except psycopg.OperationalError as err:
        raise HTTPException(status_code=503, detail="the database is not available") from err
    hits = [
        SearchHit(citation=f.citation, doc_id=f.doc_id, chunk_index=f.chunk_index, title=f.title, kind=f.kind, text=f.text, similarity=f.similarity)
        for f in found
    ]
    return SearchResponse(question=q, kind=kind, results=hits)

Step 3: dependencies

In deps.py. The model loads once (lru_cache), and the store is built per request from DATABASE_URL:

@lru_cache
def get_embedder() -> Embedder:
    return FastEmbedder()   # loads the model once, then every request reuses it


def get_store() -> Searcher:
    return ChunkStore(os.environ.get("DATABASE_URL", "postgresql://assistant:assistant@localhost:5433/assistant"))


EmbedderDep = Annotated[Embedder, Depends(get_embedder)]
StoreDep = Annotated[Searcher, Depends(get_store)]

Step 4: tests

The tests replace both dependencies, so they run in milliseconds without a model or a database:

from collections.abc import Iterator

import numpy as np
import psycopg
import pytest
from fastapi.testclient import TestClient

from my_ai_project.api import app
from my_ai_project.deps import get_embedder, get_store
from my_ai_project.embedder import FakeEmbedder
from my_ai_project.retrieval import normalize
from my_ai_project.store import ChunkRecord, Found

RECORDS = [
    ChunkRecord("error-e101", 0, "E101 Sensor timeout", "error", "Check the sensor cable and restart the monitor.", 9),
    ChunkRecord("faq-battery", 0, "Battery life", "faq", "The battery lasts about eight hours.", 7),
    ChunkRecord("manual-install", 0, "Installation guide", "manual", "Mount the sensor on a flat wall and connect the cable.", 10),
]


class InMemoryStore:
    """The same search as the database store, in memory, so the API tests need no database."""

    def __init__(self) -> None:
        self.vectors = normalize(FakeEmbedder().embed([f"{r.title}. {r.text}" for r in RECORDS]))

    def search(self, query_vector: np.ndarray, top_k: int = 3, kind: str | None = None) -> list[Found]:
        scores = self.vectors @ (query_vector / np.linalg.norm(query_vector))
        order = [i for i in scores.argsort()[::-1] if kind is None or RECORDS[i].kind == kind][:top_k]
        return [Found(RECORDS[i].doc_id, RECORDS[i].chunk_index, RECORDS[i].title, RECORDS[i].kind, RECORDS[i].text, round(float(scores[i]), 2)) for i in order]


class BrokenStore:
    def search(self, query_vector: np.ndarray, top_k: int = 3, kind: str | None = None) -> list[Found]:
        raise psycopg.OperationalError("connection refused")


@pytest.fixture
def client() -> Iterator[TestClient]:
    app.dependency_overrides[get_embedder] = FakeEmbedder
    app.dependency_overrides[get_store] = InMemoryStore
    yield TestClient(app)
    app.dependency_overrides.clear()


def test_a_search_returns_cited_chunks(client: TestClient) -> None:
    response = client.get("/api/search", params={"q": "sensor cable", "top_k": 2})
    assert response.status_code == 200
    body = response.json()
    assert body["question"] == "sensor cable"
    assert len(body["results"]) == 2
    first = body["results"][0]
    assert first["citation"] == f"[{first['doc_id']}#{first['chunk_index']}]"
    assert set(first) == {"citation", "doc_id", "chunk_index", "title", "kind", "text", "similarity"}


def test_the_kind_filter_keeps_only_that_kind(client: TestClient) -> None:
    body = client.get("/api/search", params={"q": "sensor cable", "kind": "manual"}).json()
    assert [r["doc_id"] for r in body["results"]] == ["manual-install"]
    assert body["kind"] == "manual"


@pytest.mark.parametrize(
    "params",
    [{"q": ""}, {"q": "x", "kind": "errors"}, {"q": "x", "top_k": 0}, {"q": "x", "top_k": 11}],
)
def test_bad_parameters_are_a_422(client: TestClient, params: dict[str, object]) -> None:
    assert client.get("/api/search", params=params).status_code == 422  # type: ignore[arg-type]


def test_a_database_that_is_down_is_a_503(client: TestClient) -> None:
    app.dependency_overrides[get_store] = BrokenStore
    response = client.get("/api/search", params={"q": "sensor cable"})
    assert response.status_code == 503
    assert response.json() == {"detail": "the database is not available"}

6 · Stuck? anytime

  • The first request is slow: the model loads once. The second is fast.
  • A 500 instead of a 503: the error you catch must be psycopg.OperationalError, and the store must connect inside the request.
  • Tests try to reach the database: the override of get_store is missing, or it was cleared before the request.
  • mypy complains about the fake store: its search must match the Searcher protocol.
  • An empty result list: check the kind, and check that the ingest ran after the migration.

8 · Check yourself 5 min

  1. Why are the embedder and the store dependencies and not global variables?
  2. Why is a database that is down a 503 and a wrong kind a 422?
  3. What would the citation be used for next week?
Answers
  1. A dependency can be replaced in a test, so the tests need no model and no database, and the model is loaded once and shared.
  2. A 422 means the caller sent something wrong; a 503 means the service cannot work right now, and the caller may try again later.
  3. The answer will quote a chunk and show its citation, so a reader can check where each claim came from.
