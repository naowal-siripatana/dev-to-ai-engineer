Step 1: the gateway

The client of Day 84 gets one generic PostAsync, a role header and a mapping for 422. New models, a second endpoint, and three more tests:

using System.ComponentModel.DataAnnotations;
using System.Text.Json;

namespace AskGateway;

// The shapes of the Python service. The JSON names are snake_case there (top_k, doc_id, tool_input),
// and PascalCase here: the naming policy in ApiJson converts between them.
public record AskRequest(
    [property: Required, StringLength(500, MinimumLength = 1)] string Question,
    [property: Range(1, 10)] int TopK = 3);

public record Source(string Citation, string DocId, int ChunkIndex, string Title, string Kind, string Text, double Similarity);

public record AskResponse(string Question, string Answer, bool Refused, string Reason, List<Source> Sources);

public record AssistRequest([property: Required, StringLength(500, MinimumLength = 1)] string Question);

public record ToolCallInfo(string Name, Dictionary<string, JsonElement> ToolInput, bool IsError);

public record AssistResponse(string Answer, List<ToolCallInfo> ToolCalls, int Rounds, bool GaveUp);
using System.Net;

namespace AskGateway;

public class PythonServiceException(int statusCode, string message) : Exception(message)
{
    public int StatusCode { get; } = statusCode;
}

public class PythonServiceClient(HttpClient http, ILogger<PythonServiceClient> logger)
{
    public const string RoleHeader = "x-demo-role";   // DEMO ONLY: a real role comes from a login (Module 8)

    public Task<AskResponse> AskAsync(AskRequest request, string? role, CancellationToken cancellationToken) =>
        PostAsync<AskRequest, AskResponse>("/api/ask", request, role, cancellationToken);

    public Task<AssistResponse> AssistAsync(AssistRequest request, string? role, CancellationToken cancellationToken) =>
        PostAsync<AssistRequest, AssistResponse>("/api/assist", request, role, cancellationToken);

    async Task<TResponse> PostAsync<TRequest, TResponse>(string path, TRequest request, string? role, CancellationToken cancellationToken)
    {
        using var message = new HttpRequestMessage(HttpMethod.Post, path) { Content = JsonContent.Create(request, options: ApiJson.Options) };
        if (!string.IsNullOrEmpty(role))
        {
            message.Headers.Add(RoleHeader, role);
        }

        HttpResponseMessage response;
        try
        {
            response = await http.SendAsync(message, cancellationToken);
        }
        catch (HttpRequestException error)
        {
            logger.LogWarning("python service is not reachable: {Message}", error.Message);
            throw new PythonServiceException(StatusCodes.Status502BadGateway, "the AI service is not reachable");
        }
        catch (TaskCanceledException) when (!cancellationToken.IsCancellationRequested)
        {
            logger.LogWarning("python service timed out");
            throw new PythonServiceException(StatusCodes.Status504GatewayTimeout, "the AI service took too long");
        }

        using (response)
        {
            if (response.StatusCode == HttpStatusCode.ServiceUnavailable)
            {
                throw new PythonServiceException(StatusCodes.Status503ServiceUnavailable, "the AI service cannot reach its database");
            }
            if (response.StatusCode == HttpStatusCode.UnprocessableEntity)
            {
                throw new PythonServiceException(StatusCodes.Status400BadRequest, "the AI service rejected the request (check the role)");
            }
            if (!response.IsSuccessStatusCode)
            {
                logger.LogWarning("python service answered {Status}", (int)response.StatusCode);
                throw new PythonServiceException(StatusCodes.Status502BadGateway, $"the AI service answered with status {(int)response.StatusCode}");
            }
            var answer = await response.Content.ReadFromJsonAsync<TResponse>(ApiJson.Options, cancellationToken);
            return answer ?? throw new PythonServiceException(StatusCodes.Status502BadGateway, "the AI service sent an empty answer");
        }
    }
}

Program.cs now has two endpoints that share one helper:

using AskGateway;

var builder = WebApplication.CreateBuilder(args);

builder.Services.AddValidation();                       // checks [Required], [StringLength], [Range] on the request
builder.Services.ConfigureHttpJsonOptions(o => o.SerializerOptions.PropertyNamingPolicy = ApiJson.Options.PropertyNamingPolicy);
builder.Services.AddHttpClient<PythonServiceClient>(client =>
{
    var settings = builder.Configuration.GetSection("PythonService");
    client.BaseAddress = new Uri(settings["BaseUrl"] ?? "http://localhost:8000");
    client.Timeout = TimeSpan.FromSeconds(settings.GetValue("TimeoutSeconds", 30));
});

var app = builder.Build();

app.MapGet("/health", () => Results.Ok(new { status = "ok" }));

app.MapPost("/api/ask", (AskRequest request, HttpContext http, PythonServiceClient python, CancellationToken cancellationToken) =>
    Forward(() => python.AskAsync(request, http.Request.Headers[PythonServiceClient.RoleHeader].FirstOrDefault(), cancellationToken)));

app.MapPost("/api/assist", (AssistRequest request, HttpContext http, PythonServiceClient python, CancellationToken cancellationToken) =>
    Forward(() => python.AssistAsync(request, http.Request.Headers[PythonServiceClient.RoleHeader].FirstOrDefault(), cancellationToken)));

app.Run();

static async Task<IResult> Forward<T>(Func<Task<T>> call)
{
    try
    {
        return Results.Ok(await call());
    }
    catch (PythonServiceException error)
    {
        return Results.Problem(detail: error.Message, statusCode: error.StatusCode);
    }
}

Step 3: the API client

src/api.ts: the types of the response, one function, and an ApiError that carries the status. A network failure and an error status both become an ApiError with a message a person can read:

export type Role = 'customer' | 'staff'

export interface ToolCall {
  name: string
  tool_input: Record<string, unknown>
  is_error: boolean
}

export interface AssistResponse {
  answer: string
  tool_calls: ToolCall[]
  rounds: number
  gave_up: boolean
}

export class ApiError extends Error {
  status: number

  constructor(status: number, message: string) {
    super(message)
    this.status = status
  }
}

// DEMO ONLY: the role is sent in a header so that you can try both. A real app sends a login token (Module 8).
export async function assist(question: string, role: Role): Promise<AssistResponse> {
  let response: Response
  try {
    response = await fetch('/api/assist', {
      method: 'POST',
      headers: { 'content-type': 'application/json', 'x-demo-role': role },
      body: JSON.stringify({ question }),
    })
  } catch {
    throw new ApiError(0, 'Cannot reach the server. Is the gateway running?')
  }
  if (!response.ok) {
    const problem = (await response.json().catch(() => null)) as { detail?: string; title?: string } | null
    throw new ApiError(response.status, problem?.detail ?? problem?.title ?? `The request failed (${response.status}).`)
  }
  return (await response.json()) as AssistResponse
}

Step 4: the chat component

src/App.tsx. The state is a list of messages, the text box, the role, and a waiting flag that disables the button while a call is running. The tool calls of an answer are in a details element, so they are there when you want them and out of the way when you do not:

import { useState, type FormEvent } from 'react'
import { ApiError, assist, type Role, type ToolCall } from './api'
import './index.css'

interface Message {
  id: number
  from: 'user' | 'assistant'
  text: string
  toolCalls?: ToolCall[]
  failed?: boolean
}

export default function App() {
  const [messages, setMessages] = useState<Message[]>([])
  const [question, setQuestion] = useState('')
  const [role, setRole] = useState<Role>('customer')
  const [waiting, setWaiting] = useState(false)

  function add(message: Omit<Message, 'id'>) {
    setMessages((current) => [...current, { ...message, id: current.length }])
  }

  async function send(event: FormEvent) {
    event.preventDefault()
    const text = question.trim()
    if (!text || waiting) return
    add({ from: 'user', text })
    setQuestion('')
    setWaiting(true)
    try {
      const result = await assist(text, role)
      add({ from: 'assistant', text: result.answer, toolCalls: result.tool_calls })
    } catch (error) {
      const message = error instanceof ApiError ? error.message : 'Something went wrong.'
      add({ from: 'assistant', text: message, failed: true })
    } finally {
      setWaiting(false)
    }
  }

  return (
    <main>
      <header>
        <h1>PumpGuard assistant</h1>
        <label>
          I am a{' '}
          <select value={role} onChange={(event) => setRole(event.target.value as Role)}>
            <option value="customer">customer</option>
            <option value="staff">staff member</option>
          </select>
        </label>
      </header>

      <section aria-live="polite" aria-label="Conversation">
        {messages.length === 0 && <p className="hint">Ask about an error code, for example: What does error E104 mean?</p>}
        {messages.map((message) => (
          <article key={message.id} className={`${message.from}${message.failed ? ' failed' : ''}`}>
            <p>{message.text}</p>
            {message.toolCalls && message.toolCalls.length > 0 && (
              <details>
                <summary>Tool calls ({message.toolCalls.length})</summary>
                <ul>
                  {message.toolCalls.map((call, index) => (
                    <li key={index}>
                      <code>
                        {call.name}({JSON.stringify(call.tool_input)})
                      </code>{' '}
                      {call.is_error ? 'failed' : 'ok'}
                    </li>
                  ))}
                </ul>
              </details>
            )}
          </article>
        ))}
        {waiting && <p className="hint">Thinking...</p>}
      </section>

      <form onSubmit={send}>
        <input value={question} onChange={(event) => setQuestion(event.target.value)} placeholder="Type a question" aria-label="Your question" maxLength={500} />
        <button type="submit" disabled={waiting || question.trim() === ''}>
          Send
        </button>
      </form>
    </main>
  )
}

6 · Stuck? anytime

  • The page shows “Cannot reach the server”: the gateway is not running on 5100, or the proxy in vite.config.ts has another port. Test with the curl command first.
  • A 404 from Vite on /api/assist: restart npm run dev after changing vite.config.ts.
  • A 400 from the gateway: the request body is {"question": "..."} and the question must not be empty or longer than 500 characters.
  • A 502 although Python runs: the gateway looks at PythonService:BaseUrl (port 8000).
  • TypeScript errors about type imports: with the Vite template, import types with import { type Role } or import type.
  • You want a real model: start the Python service with LLM_MODE=real and ANTHROPIC_API_KEY. The page does not change.

8 · Check yourself 5 min

  1. Why does the page call the gateway and not the Python service?
  2. Why does the Vite proxy mean you need no CORS during development?
  3. Why is the role selector only a demo?
Answers
  1. The .NET layer is the front door: it will own login, business rules and logging. The browser should know one address, and the AI service stays behind it.
  2. The browser only talks to its own origin (port 5173). Vite forwards the request to the gateway from the server side, where CORS does not apply.
  3. Anything in the browser can be changed by the user, so a role that the page sends can be faked. A real role comes from a login that the server checks (Module 8).
